\section{Bắt đầu nhanh}

\begin{frame}[fragile]{File tối thiểu}
Một bài trình chiếu mới chỉ cần chọn lớp \texttt{beamer}, nạp theme HUST và viết nội dung trong \texttt{frame}.

\begin{minted}{latex}
\documentclass[aspectratio=169]{beamer}
\usetheme[theme=blue,logo=hustwithtext,assetext=pdf]{HUST}

\title{Tên bài trình bày}
\author{Tên tác giả}

\begin{document}
\HUSTInsertBrandSlide
\begin{frame}{Tiêu đề slide}
  Nội dung chính của slide.
% ... đóng frame ở đây
\end{document}
\end{minted}
\end{frame}

\begin{frame}[fragile]{Build bằng latexmk}
\texttt{latexmkrc} trong thư mục này đã cấu hình lệnh build chính cho template.

\begin{minted}{bash}
latexmk main.tex
\end{minted}

\begin{block}{Ghi nhớ}
Template đang dùng \texttt{minted}, vì vậy quy trình build cần hỗ trợ \texttt{-shell-escape}. Cấu hình này đã nằm trong \texttt{latexmkrc}; người dùng thường chỉ cần chạy \texttt{latexmk main.tex}.
\end{block}
\end{frame}

\section{Tùy chọn template}

\begin{frame}[fragile]{Chọn màu, logo và định dạng asset}
Các tùy chọn quan trọng nằm ngay trong dòng \texttt{\textbackslash usetheme}.

\begin{minted}{latex}
\usetheme[
  theme=blue,          % blue hoặc red
  logo=hustwithtext,   % biến thể logo ở footer
  assetext=pdf         % pdf, png hoặc jpg
]{HUST}
\end{minted}

\begin{itemize}
  \item \texttt{theme} đổi hệ màu và thư mục nền tương ứng.
  \item \texttt{logo} chọn logo xuất hiện ở thanh cuối slide.
  \item \texttt{assetext} quyết định phần mở rộng mặc định cho ảnh nền.
\end{itemize}
\end{frame}

\begin{frame}[fragile]{Tỷ lệ slide}
Tỷ lệ slide được chọn ở \texttt{documentclass}; theme sẽ tự tìm đúng thư mục asset.

\begin{columns}[T]
  \begin{column}{0.48\textwidth}
    \textbf{Màn hình rộng}
    \begin{minted}{latex}
\documentclass[
  aspectratio=169
]{beamer}
    \end{minted}
  \end{column}
  \begin{column}{0.48\textwidth}
    \textbf{Tỷ lệ 4:3}
    \begin{minted}{latex}
\documentclass[
  aspectratio=43
]{beamer}
    \end{minted}
  \end{column}
\end{columns}

\vspace{0.5em}
Với \texttt{theme=blue} và \texttt{aspectratio=169}, nền mặc định được đọc từ \texttt{assets/blue\_16x9/}.
\end{frame}

\begin{frame}[fragile]{Khi nào dùng PDF, PNG hoặc JPG?}
\begin{itemize}
  \item \texttt{assetext=pdf}: phù hợp nhất cho logo, nền vector và in ấn sắc nét.
  \item \texttt{assetext=png}: phù hợp cho ảnh raster cần nền trong suốt.
  \item \texttt{assetext=jpg}: phù hợp cho ảnh chụp hoặc nền raster lớn không cần trong suốt.
\end{itemize}

\begin{minted}{latex}
% Dùng PDF cho toàn bộ nền không ghi phần mở rộng.
\usetheme[assetext=pdf]{HUST}
\HUSTUseBackground{theme}

% Ghi đè từng slide bằng phần mở rộng cụ thể.
\HUSTUseBackground{theme.png}
\end{minted}
\end{frame}

\section{Nền và slide thương hiệu}

\begin{frame}[fragile]{Các slide nền có sẵn}
Các lệnh chèn slide thương hiệu đã được đóng gói sẵn để background không rò sang slide tiếp theo.

\begin{minted}{latex}
\HUSTInsertBrandSlide
\HUSTInsertThemeSlide
\HUSTInsertOnesideSlide
\end{minted}

\begin{itemize}
  \item \texttt{BrandSlide}: slide thương hiệu đầy đủ.
  \item \texttt{ThemeSlide}: slide chuyển phần hoặc mở đầu chủ đề.
  \item \texttt{OnesideSlide}: nền một phía, tiện cho slide cảm ơn hoặc thông báo.
\end{itemize}
\end{frame}

\begin{frame}[fragile]{Đổi nền cho một nhóm slide}
Dùng cặp ngoặc nhọn để giới hạn phạm vi background cho một slide hoặc một nhóm slide.

\begin{minted}{latex}
{\HUSTUseBackground{theme_hust_oneside}
\begin{frame}{Thông báo quan trọng}
  Nội dung đặt trên nền riêng.
% ... đóng frame ở đây
}
\end{minted}

\begin{block}{Thói quen nên dùng}
Luôn giới hạn phạm vi khi chỉ muốn đổi nền tạm thời. Nếu cần xóa nền thủ công, dùng \texttt{\textbackslash HUSTClearBackground}.
\end{block}
\end{frame}

{\HUSTUseBackground{theme_hust_oneside}
\begin{frame}{Ví dụ nền riêng}
  \begin{columns}[T]
    \begin{column}{0.47\textwidth}

    \end{column}
    \begin{column}{0.47\textwidth}
      Vì lệnh nằm trong một nhóm, các slide sau quay lại nền thông thường.
    \end{column}
  \end{columns}
\end{frame}
}

\section{Hoàn thiện bài trình chiếu}

\begin{frame}{Checklist trước khi build}
\begin{itemize}
  \item Chọn \texttt{aspectratio=169} hoặc \texttt{aspectratio=43} ngay từ đầu.
  \item Chọn \texttt{theme=blue} hoặc \texttt{theme=red} theo bối cảnh sử dụng.
  \item Chọn \texttt{logo} đúng đơn vị hoặc ngôn ngữ.
  \item Chọn \texttt{assetext=pdf} nếu ưu tiên chất lượng vector.
  \item Kiểm tra các slide có nền riêng để tránh background rò sang slide sau.
\end{itemize}
\end{frame}
