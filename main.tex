\section{Definition box}

% Cách 1: sử dụng môi trường definition
\begin{frame}{Definition box với môi trường \texttt{definition}}
\begin{definition}[Algorithm]
An \alert{algorithm} is a finite sequence of well-defined instructions, typically used to solve a class of specific problems or perform a computation.
\end{definition}

\vspace{0.7em}
Gợi ý:
\begin{itemize}
  \item Bạn có thể đổi tên môi trường (\texttt{Definition} $\to$ \texttt{Định nghĩa}) bằng cách chỉnh lại \texttt{\textbackslash newtheorem}.
  \item Có thể dùng thêm \texttt{\textbackslash alert\{\}} để nhấn mạnh.
\end{itemize}
\end{frame}

% Cách 2: dùng block (nếu muốn style đồng bộ với theme)
\begin{frame}{Definition box với \texttt{block}}
\begin{block}{Definition: Complexity}
The \emph{time complexity} of an algorithm describes how the runtime grows with the size of the input.
\end{block}

\begin{exampleblock}{Example}
The time complexity of binary search on a sorted array of size $n$ is $O(\log n)$.
\end{exampleblock}
\end{frame}

\section{Chèn code}

\begin{frame}[fragile]{Chèn code với gói \texttt{minted}}
% [fragile] là bắt buộc khi frame có môi trường verbatim

Ví dụ đoạn code Python:

\begin{minted}[fontsize=\small, breaklines, autogobble]{python}
def fib(n):
  # Cấu trúc dữ liệu và Thuật toán
  a, b = 0, 1
  for _ in range(n):
    a, b = b, a + b
  return a
print("hello world ", fib(20))
\end{minted}

Gợi ý:
\begin{itemize}
  \item Thêm thuộc tính \texttt{language=Python}, \texttt{language=Java}, v.v. cho các ngôn ngữ khác.
  \item Có thể định nghĩa \texttt{\textbackslash lstdefinestyle} để đồng bộ style trong toàn bộ slide.
\end{itemize}
\end{frame}

\begin{frame}[fragile]{Chèn code từ file}

\end{frame}
